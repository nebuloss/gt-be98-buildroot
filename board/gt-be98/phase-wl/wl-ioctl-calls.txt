# TRANSPORT (all rows): ioctl(SIOCDEVPRIVATE=0x89F0) on AF_INET/SOCK_DGRAM socket,
#   ifr.ifr_name=<netdev = radio "wlX" or BSS "wlX.Y">, ifr.ifr_data=&wl_ioctl_t.
#   wl_ioctl_t { u32 cmd; void* buf; u32 len; u8 set; u32 used; u32 needed; }  (24B, 32-bit ARM)
#   WLC_IOCTL_MAGIC=0x14e46c77  WLC_IOCTL_MAXLEN=16384  WLC_GET_MAGIC=cmd0 (presence probe)
#   iovars: cmd=WLC_GET_VAR(262)/WLC_SET_VAR(263); buf = "name\0"[+payload].
#
# DERIVATION (read-only):
#   struct/magic/numbers:
#     grep -n 'wl_ioctl\|WLC_IOCTL_MAGIC\|FOREACH_WLIOCTL' \
#       <fw>/bcmdrivers/broadcom/net/wl/impl105/main/components/wlioctl/include/{wlioctl.h,wlioctl_defs.h}
#   transport:  read <fw>/release/src/router/shared/wl_linux.c (wl_ioctl)
#   iovar wrappers: grep wl_iovar_ <fw>/release/src/router/shared/wlutils.h
#   rc usage:   grep -roE 'eval\("wl"...|wl_iovar_(set|get)int?\(...,"name"' <fw>/release/src/router/rc/*.c
#   binary:     readelf -d / strings  /usr/sbin/wl  (0031 rootfs; links only librt/libc/ld)
#   live:       ssh admin@10.0.0.8 -p2222  'wl [-i IF] {ver,status,bss,ssid,radio,chanspec,isup,assoclist}'  (GET only)
#
